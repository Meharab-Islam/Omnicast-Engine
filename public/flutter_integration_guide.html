সার্ভার কানেকশন (REST Auth & WebSocket)

ক্লায়েন্ট SDK প্রথমে POST /api/auth/token-এ কল করে signed JWT টোকেন এবং Coturn STUN/TURN ক্রেডেনশিয়াল সংগ্রহ করে। এরপর সেই টোকেন দিয়ে WebSocket কানেকশন তৈরি করে।

auth_and_signaling.dartDart Implementation
import 'dart:convert';
import 'package:http/http.dart' as http;
import 'package:web_socket_channel/web_socket_channel.dart';
import 'package:web_socket_channel/io.dart';

class AuthResponse {
  final String token;
  final List<Map<String, dynamic>> iceServers;

  AuthResponse({required this.token, required this.iceServers});
}

/// ১. REST API থেকে JWT এবং ডাইনামিক Coturn ICE সার্ভার সংগ্রহ করা
Future<AuthResponse> authenticateSDK({
  required String hostUrl,
  required String apiKey,
  required String apiSecret,
  required String userId,
  String userName = 'Flutter User',
}) async {
  final url = Uri.parse('$hostUrl/api/auth/token');
  final res = await http.post(
    url,
    headers: {'Content-Type': 'application/json'},
    body: jsonEncode({
      'api_key': apiKey,
      'api_secret': apiSecret,
      'user_id': userId,
      'user_name': userName,
    }),
  );

  if (res.statusCode != 200) {
    throw Exception('Auth failed: ${res.body}');
  }

  final data = jsonDecode(res.body);
  final rawIce = (data['ice_servers'] as List? ?? []);
  final iceServers = rawIce.map((e) => Map<String, dynamic>.from(e)).toList();

  return AuthResponse(token: data['token'], iceServers: iceServers);
}

/// ২. WebSocketChannel কানেক্ট করা
WebSocketChannel connectSignaling({
  required String hostUrl,
  required String token,
}) {
  final wsProtocol = hostUrl.startsWith('https') ? 'wss' : 'ws';
  final cleanHost = hostUrl.replaceAll(RegExp(r'^https?://'), '').replaceAll('/', '');
  final wsUri = Uri.parse('$wsProtocol://$cleanHost/ws?token=$token');

  return IOWebSocketChannel.connect(wsUri);
}

video_publisher.dartSimulcast Publishing Implementation
import 'dart:convert';
import 'package:flutter_webrtc/flutter_webrtc.dart';
import 'package:web_socket_channel/web_socket_channel.dart';

class LiveHostEngine {
  final WebSocketChannel channel;
  final List<Map<String, dynamic>> iceServers;
  final String roomId;
  final String userId;

  RTCPeerConnection? peerConnection;
  MediaStream? localStream;
  final RTCVideoRenderer localRenderer = RTCVideoRenderer();

  LiveHostEngine({
    required this.channel,
    required this.iceServers,
    required this.roomId,
    required this.userId,
  });

  Future<void> startBroadcast() async {
    // ১. RTCVideoRenderer ইনিশিয়ালাইজ করা
    await localRenderer.initialize();

    // ২. ক্যামেরা ও মাইক থেকে মিডিয়া স্ট্রিম সংগ্রহ
    localStream = await navigator.mediaDevices.getUserMedia({
      'audio': true,
      'video': {
        'facingMode': 'user',
        'width': 1280,
        'height': 720,
      }
    });
    localRenderer.srcObject = localStream;

    // ৩. PeerConnection তৈরি করা
    final config = {
      'iceServers': iceServers,
      'sdpSemantics': 'unified-plan',
    };
    peerConnection = await createPeerConnection(config);

    // ICE ক্যান্ডিডেট সার্ভারে পাঠানো
    peerConnection!.onIceCandidate = (RTCIceCandidate candidate) {
      if (candidate.candidate != null) {
        channel.sink.add(jsonEncode({
          'action': 'ice',
          'room_id': roomId,
          'user_id': userId,
          'payload': candidate.toMap(),
        }));
      }
    };

    // ৪. CRITICAL: ৩টি সিমুলকাস্ট লেয়ার (f, h, q) সহ Transceiver যোগ করা
    final videoTrack = localStream!.getVideoTracks().first;
    await peerConnection!.addTransceiver(
      track: videoTrack,
      init: RTCRtpTransceiverInit(
        direction: TransceiverDirection.SendOnly,
        streams: [localStream!],
        sendEncodings: [
          RTCRtpEncoding(rid: 'f', active: true, maxBitrate: 1200000),                         // High (720p/1080p)
          RTCRtpEncoding(rid: 'h', active: true, maxBitrate: 500000, scaleResolutionDownBy: 2.0), // Medium (480p)
          RTCRtpEncoding(rid: 'q', active: true, maxBitrate: 150000, scaleResolutionDownBy: 4.0), // Low (240p)
        ],
      ),
    );

    // অডিও ট্রান্সসিভার যোগ করা
    final audioTrack = localStream!.getAudioTracks().first;
    await peerConnection!.addTransceiver(
      track: audioTrack,
      init: RTCRtpTransceiverInit(
        direction: TransceiverDirection.SendOnly,
        streams: [localStream!],
      ),
    );

    // ৫. SDP Offer তৈরি ও লোকাল ডেসক্রিপশন সেট করা
    final offer = await peerConnection!.createOffer();
    await peerConnection!.setLocalDescription(offer);

    // ৬. সার্ভারে publish অফার মেসেজ পাঠানো
    channel.sink.add(jsonEncode({
      'action': 'publish',
      'room_id': roomId,
      'user_id': userId,
      'payload': {
        'sdp': offer.sdp,
        'type': offer.type,
      }
    }));
  }
}

class MediaController {
  final MediaStream? localStream;
  final WebSocketChannel channel;
  final String roomId;
  final String userId;

  bool isMicMuted = false;
  bool isCameraMuted = false;

  MediaController({
    required this.localStream,
    required this.channel,
    required this.roomId,
    required this.userId,
  });

  /// মাইক্রোফোন মিউট/আনমিউট টগল
  void toggleMicrophone() {
    if (localStream == null) return;
    final audioTrack = localStream!.getAudioTracks().firstOrNull;
    if (audioTrack != null) {
      isMicMuted = !isMicMuted;
      audioTrack.enabled = !isMicMuted; // WebRTC অডিও ফ্রেম পাঠানো বন্ধ/চালু

      // সার্ভারে মিডিয়া স্টেট আপডেট পাঠানো
      channel.sink.add(jsonEncode({
        'action': 'media_state',
        'room_id': roomId,
        'user_id': userId,
        'payload': {
          'muted_audio': isMicMuted,
          'muted_video': isCameraMuted,
        }
      }));
    }
  }
}

সিটে ওঠা ও কো-হোস্টিং (Seat Management & Dynamic Renegotiation)

কো-হোস্টিং সিস্টেমে যখন একজন দর্শক সিটে ওঠে, তখন সার্ভার সমস্ত দর্শকের কাছে একটি নতুন SDP Offer পাঠায়। ক্লায়েন্ট SDK সেই অফারটি setRemoteDescription-এ সেট করে এবং নতুন কো-হোস্টের ভিডিও ফ্রেম স্ক্রিনে রেন্ডার করে।

cohost_and_renegotiation.dartViewer Dynamic Track Handling
class LiveViewerEngine {
  final WebSocketChannel channel;
  RTCPeerConnection? peerConnection;
  final RTCVideoRenderer mainHostRenderer = RTCVideoRenderer();
  final Map<String, RTCVideoRenderer> coHostRenderers = {};

  /// দর্শক সিটের জন্য রিকোয়েস্ট পাঠানো
  void requestSeat(int seatId, String roomId, String userId) {
    channel.sink.add(jsonEncode({
      'action': 'seat_request',
      'room_id': roomId,
      'user_id': userId,
      'payload': {'seat_id': '$seatId'}
    }));
  }

  /// WebSocket থেকে ইনকামিং সিগন্যালিং মেসেজ হ্যান্ডেল করা
  Future<void> handleSignalingMessage(Map<String, dynamic> msg) async {
    final action = msg['action'] ?? msg['event'];
    final payload = msg['payload'] ?? msg['data'] ?? {};

    // ১. ডাইনামিক Renegotiation অফার আসলে (কো-হোস্ট সিটে যুক্ত হলে)
    if (action == 'offer' || action == 'renegotiate') {
      final sdp = payload['sdp'];
      final type = payload['type'] ?? 'offer';

      await peerConnection!.setRemoteDescription(RTCSessionDescription(sdp, type));
      final answer = await peerConnection!.createAnswer();
      await peerConnection!.setLocalDescription(answer);

      channel.sink.add(jsonEncode({
        'action': 'answer',
        'payload': {'sdp': answer.sdp, 'type': answer.type}
      }));
    }

    // ২. সিট খালি হলে কো-হোস্ট রেন্ডারার ক্লিনআপ
    if (action == 'seat_left' || action == 'cohost_left') {
      final userId = payload['user_id'];
      if (coHostRenderers.containsKey(userId)) {
        coHostRenderers[userId]!.srcObject = null;
        coHostRenderers.remove(userId);
      }
    }
  }

  /// WebRTC OnTrack লিসেনার: হোস্ট এবং কো-হোস্টের ট্র্যাক ডাইনামিকভাবে বাইন্ড করা
  void setupTrackListeners() {
    peerConnection!.onTrack = (RTCTrackEvent event) async {
      if (event.track.kind == 'video') {
        if (mainHostRenderer.srcObject == null) {
          mainHostRenderer.srcObject = event.streams.first;
        } else {
          // নতুন কো-হোস্টের জন্য ডাইনামিক রেন্ডারার তৈরি
          final coHostRenderer = RTCVideoRenderer();
          await coHostRenderer.initialize();
          coHostRenderer.srcObject = event.streams.first;
          coHostRenderers[event.track.id!] = coHostRenderer;
        }
      }
    };
  }
}

class PKBattleSplitView extends StatelessWidget {

  final RTCVideoRenderer host2Renderer;

            children: [
              Expanded(
                flex: (ratio1 * 100).toInt().clamp(5, 95),
                child: Container(color: Colors.redAccent),

              Expanded(
                flex: ((1 - ratio1) * 100).toInt().clamp(5, 95),

live_state_manager.dartChangeNotifier State Store
import 'dart:convert';
import 'package:flutter/foundation.dart';
import 'package:web_socket_channel/web_socket_channel.dart';

class LiveRoomState extends ChangeNotifier {
  String roomId = '';
  String hostId = '';
  int totalViewers = 0;
  int hostScore = 0;
  Map<String, String> activeSeats = {};
  final List<Map<String, dynamic>> chatMessages = [];

  /// WebSocket স্ট্রিম লিসেন করা
  void bindSignalingStream(WebSocketChannel channel) {
    channel.stream.listen((rawData) {
      try {
        final msg = jsonDecode(rawData as String);
        final action = msg['action'] ?? msg['event'];
        final payload = msg['payload'] ?? msg['data'] ?? {};

        // ১. লেট জয়েনার স্ন্যাপশট দিয়ে স্টেট পূরণ
        if (action == 'room_info_sync' || action == 'room_state') {
          roomId = payload['room_id'] ?? roomId;
          hostId = payload['host_id'] ?? hostId;
          totalViewers = payload['total_viewers'] ?? 0;
          hostScore = payload['host_score'] ?? 0;
          activeSeats = Map<String, String>.from(payload['active_seats'] ?? {});
          notifyListeners();
        }

        // ২. গিফট প্রসেস হলে রিয়েল-টাইম ডায়মন্ড স্কোর আপডেট
        if (action == 'gift_processed' || action == 'score_updated') {
          if (payload['new_score'] != null) {
            hostScore = payload['new_score'];
            notifyListeners();
          }
        }

        // ৩. ইনকামিং চ্যাট মেসেজ লিস্টে যুক্ত করা
        if (action == 'chat') {
          chatMessages.add(payload);
          notifyListeners();
        }
      } catch (e) {
        // Parse error
      }
    });
  }

  /// গিফট পাঠানো
  void sendGift(WebSocketChannel channel, String giftName, int coins, String userId) {
    channel.sink.add(jsonEncode({
      'action': 'gift',
      'room_id': roomId,
      'user_id': userId,
      'payload': {
        'gift_name': giftName,
        'coins': coins,
      }
    }));
  }
}
